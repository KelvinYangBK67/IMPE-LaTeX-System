% Feature module: citations

\ifdefined\NextFeatureCitationsLoaded
  \endinput
\fi
\def\NextFeatureCitationsLoaded{1}

\RequirePackage{csquotes}

\providecommand{\NextCitationBiblatexOptions}{%
  backend=biber,
  style=apa,
  sorting=nyt,
  sortcites=true,
  doi=true,
  isbn=false,
  url=true
}

\makeatletter
\newif\ifNextCitationBiblatexPreloaded
\@ifpackageloaded{biblatex}
  {\NextCitationBiblatexPreloadedtrue
   \PackageWarningNoLine{feature-citations}{biblatex was already loaded; retaining its style and options. Set citation style before biblatex loading}}%
  {\NextCitationBiblatexPreloadedfalse%
    \edef\NextCitationLoadBiblatex{%
      \noexpand\RequirePackage[\NextCitationBiblatexOptions]{biblatex}%
    }%
    \NextCitationLoadBiblatex
  }

\makeatother

% Preserve already-loaded biblatex styles, including their delimiter rules.
% The historical IMPE delimiter override is retained only for IMPE-managed
% loading, so existing output does not silently change in 1.x.
\ifNextCitationBiblatexPreloaded\else
\DeclareDelimFormat{finalnamedelim}{%
  \ifnumgreater{\value{liststop}}{2}{\finalandcomma}{}%
  \addspace\&\space}
\DeclareDelimFormat[textcite]{finalnamedelim}{%
  \ifnumgreater{\value{liststop}}{2}{\finalandcomma}{}%
  \addspace\&\space}
\fi
