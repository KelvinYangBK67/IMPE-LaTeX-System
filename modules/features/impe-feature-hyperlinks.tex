% All IMPE destination identities come from this document-wide sequence.
\newcount\NextHyperDestinationSequence
\newcommand{\NextHyperlinksAllocateDestination}[1]{%
  \global\advance\NextHyperDestinationSequence by 1\relax
  \edef#1{impe.dest.\the\NextHyperDestinationSequence}%
}
\newcommand{\NextHyperlinksCurrentTocTarget}{}
\newif\ifNextHyperlinksExplicitLinkTarget
\makeatletter
% Keep explicit names supplied via MakeLinkTarget* untouched. The hook below
% only rewrites automatically allocated names, retaining their semantic type.
\NewCommandCopy{\NextHyperlinksOriginalMakeLinkTarget}{\MakeLinkTarget}
\RenewDocumentCommand{\MakeLinkTarget}{s O{} m}{%
  \IfBooleanTF{#1}{%
    \begingroup
      \NextHyperlinksExplicitLinkTargettrue
      \NextHyperlinksOriginalMakeLinkTarget*[#2]{#3}%
    \endgroup
  }{\NextHyperlinksOriginalMakeLinkTarget[#2]{#3}}%
}
\def\NextHyperlinksTypeBeforeDot#1.#2\@nil{#1}
\def\NextHyperlinksTypeWithoutStar#1*#2\@nil{#1}
\makeatother
\newif\ifNextHyperlinksHeadingShouldLink
\makeatletter
% Preserve reference types (section, figure, equation, etc.) so autoref and
% other native reference tools retain their semantics. Explicit link targets
% supplied via MakeLinkTarget* are not replaced.
\AddToHook{__hyp/target/setname}[impe]{%
  \ifNextHyperlinksExplicitLinkTarget\else
    \edef\NextHyperlinksTargetType{%
      \expandafter\NextHyperlinksTypeBeforeDot\@currentHref.\@nil}%
    % Hyperref also creates section*/chapter* targets. They are structural
    % auto-targets, but autoref expects the unstarred semantic type.
    \edef\NextHyperlinksTargetType{%
      \expandafter\NextHyperlinksTypeWithoutStar\NextHyperlinksTargetType*\@nil}%
    \NextHyperlinksAllocateDestination\NextHyperlinksAutoTarget
    \edef\NextHyperlinksAutoTarget{%
      \NextHyperlinksTargetType.impe.\the\NextHyperDestinationSequence}%
    \Hy@MakeCurrentHref{\NextHyperlinksAutoTarget}%
  \fi
}
\DeclareRobustCommand{\NextHyperlinksTocAnchor}[1]{%
  \Hy@raisedlink{\hypertarget{#1}{}}%
}
\DeclareRobustCommand{\NextHyperlinksLinkedHeading}[2]{%
  \texorpdfstring{\hyper@linkstart{link}{#1}#2\hyper@linkend}{#2}%
}

\let\NextHyperlinksOriginalAddContentsLine\addcontentsline
\renewcommand{\addcontentsline}[3]{%
  \ifx\NextHyperlinksCurrentTocTarget\@empty
    \NextHyperlinksOriginalAddContentsLine{#1}{#2}{#3}%
  \else
    \def\NextHyperlinksTocFile{#1}%
    \def\NextHyperlinksTocName{toc}%
    \ifx\NextHyperlinksTocFile\NextHyperlinksTocName
      \edef\NextHyperlinksTocTargetName{\NextHyperlinksCurrentTocTarget}%
      \addtocontents{toc}{%
        \protect\NextHyperlinksTocAnchor{\NextHyperlinksTocTargetName}%
      }%
      \NextHyperlinksOriginalAddContentsLine{#1}{#2}{#3}%
    \else
      \NextHyperlinksOriginalAddContentsLine{#1}{#2}{#3}%
    \fi
  \fi
}

\newcommand{\NextHyperlinksBeginHeadingLink}{%
  \NextHyperlinksAllocateDestination\NextHyperlinksCurrentTocTarget
  \global\NextHyperlinksHeadingShouldLinktrue
}
\newcommand{\NextHyperlinksEndHeadingLink}{%
  \global\NextHyperlinksHeadingShouldLinkfalse
  \global\let\NextHyperlinksCurrentTocTarget\@empty
}
\newcommand{\NextHyperlinksDisplayHeading}[1]{%
  \ifNextHyperlinksHeadingShouldLink
    \NextHyperlinksLinkedHeading{\NextHyperlinksCurrentTocTarget}{#1}%
  \else
    #1%
  \fi
}

\newcommand{\NextHyperlinksPatchHeadingCommands}{%
  \@ifundefined{NextHyperlinksHeadingCommandsPatched}{%
    \gdef\NextHyperlinksHeadingCommandsPatched{1}%
    \@ifundefined{@makechapterhead}{}{%
      \let\NextHyperlinksOriginalMakeChapterHead\@makechapterhead
      \def\@makechapterhead##1{%
        \NextHyperlinksOriginalMakeChapterHead{\NextHyperlinksDisplayHeading{##1}}%
      }%
    }%
    \@ifundefined{@sect}{}{%
      \let\NextHyperlinksOriginalSect\@sect
      \def\@sect##1##2##3##4##5##6[##7]##8{%
        \NextHyperlinksOriginalSect{##1}{##2}{##3}{##4}{##5}{##6}[##7]{%
          \NextHyperlinksDisplayHeading{##8}%
        }%
      }%
    }%
    \@ifundefined{chapter}{}{%
      \let\NextHyperlinksOriginalChapter\chapter
      \RenewDocumentCommand{\chapter}{s o +m}{%
        \IfBooleanTF{##1}{%
          \NextHyperlinksOriginalChapter*{##3}%
          \@ifundefined{NextUiZhMaybeAddContents}{}{%
            \IfNoValueTF{##2}
              {\NextUiZhMaybeAddContents{chapter}{##3}}
              {\NextUiZhMaybeAddContents{chapter}{##2}}%
          }%
        }{%
          \NextHyperlinksBeginHeadingLink
          \IfNoValueTF{##2}
            {\NextHyperlinksOriginalChapter[##3]{##3}}
            {\NextHyperlinksOriginalChapter[##2]{##3}}%
          \NextHyperlinksEndHeadingLink
        }%
      }%
    }%
    \let\NextHyperlinksOriginalSection\section
    \RenewDocumentCommand{\section}{s o +m}{%
      \IfBooleanTF{##1}{%
        \NextHyperlinksOriginalSection*{##3}%
        \@ifundefined{NextUiZhMaybeAddContents}{}{%
          \IfNoValueTF{##2}
            {\NextUiZhMaybeAddContents{section}{##3}}
            {\NextUiZhMaybeAddContents{section}{##2}}%
        }%
      }{%
        \NextHyperlinksBeginHeadingLink
        \IfNoValueTF{##2}
          {\NextHyperlinksOriginalSection[##3]{##3}}
          {\NextHyperlinksOriginalSection[##2]{##3}}%
        \NextHyperlinksEndHeadingLink
      }%
    }%
    \let\NextHyperlinksOriginalSubsection\subsection
    \RenewDocumentCommand{\subsection}{s o +m}{%
      \IfBooleanTF{##1}{%
        \NextHyperlinksOriginalSubsection*{##3}%
      }{%
        \NextHyperlinksBeginHeadingLink
        \IfNoValueTF{##2}
          {\NextHyperlinksOriginalSubsection[##3]{##3}}
          {\NextHyperlinksOriginalSubsection[##2]{##3}}%
        \NextHyperlinksEndHeadingLink
      }%
    }%
    \let\NextHyperlinksOriginalSubsubsection\subsubsection
    \RenewDocumentCommand{\subsubsection}{s o +m}{%
      \IfBooleanTF{##1}{%
        \NextHyperlinksOriginalSubsubsection*{##3}%
      }{%
        \NextHyperlinksBeginHeadingLink
        \IfNoValueTF{##2}
          {\NextHyperlinksOriginalSubsubsection[##3]{##3}}
          {\NextHyperlinksOriginalSubsubsection[##2]{##3}}%
        \NextHyperlinksEndHeadingLink
      }%
    }%
    \let\NextHyperlinksOriginalParagraph\paragraph
    \RenewDocumentCommand{\paragraph}{s o +m}{%
      \IfBooleanTF{##1}{%
        \NextHyperlinksOriginalParagraph*{##3}%
      }{%
        \NextHyperlinksBeginHeadingLink

          {\NextHyperlinksOriginalParagraph[##3]{##3}}
          {\NextHyperlinksOriginalParagraph[##2]{##3}}%
        \NextHyperlinksEndHeadingLink
      }%
    }%
    \let\NextHyperlinksOriginalSubparagraph\subparagraph
    \RenewDocumentCommand{\subparagraph}{s o +m}{%
      \IfBooleanTF{##1}{%

  }{}%

% Beamer footnotes have overlay and frame options; retain their native
% parser and navigation, rather than replacing it with the paper parser.
\@ifclassloaded{beamer}{}{%
\RenewDocumentCommand{\footnote}{o +m}{%
  \NextHyperlinksAllocateDestination\NextHyperFootnoteMarkTarget
  \NextHyperlinksAllocateDestination\NextHyperFootnoteTextTarget
  \begingroup
    \let\NextHyperlinksOriginalMakeFnMark\@makefnmark
    \renewcommand{\@makefnmark}{%
      \Hy@raisedlink{\hypertarget{\NextHyperFootnoteMarkTarget}{}}%
      \hyperlink{\NextHyperFootnoteTextTarget}{\NextHyperlinksOriginalMakeFnMark}%
    }%
    \IfNoValueTF{#1}
      {%
        \refstepcounter\@mpfn
        \protected@xdef\@thefnmark{\thempfn}%
      }
      {%
        \begingroup
          \csname c@\@mpfn\endcsname #1\relax
          \unrestored@protected@xdef\@thefnmark{\thempfn}%
        \endgroup
      }%
    \@footnotemark
    \renewcommand{\@makefnmark}{%
      \hyperlink{\NextHyperFootnoteMarkTarget}{\NextHyperlinksOriginalMakeFnMark}%
    }%
    \@footnotetext{%
      \Hy@raisedlink{\hypertarget{\NextHyperFootnoteTextTarget}{}}#2%
    }%
  \endgroup
}
}% End non-Beamer footnote specialization.
