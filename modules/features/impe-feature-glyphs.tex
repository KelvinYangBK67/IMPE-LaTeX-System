\RequirePackage{xsr-vector}
\RequirePackage{adjustbox}
\RequirePackage{xparse}
\ExplSyntaxOn

% Separate source identity from optional human-readable metadata.  Metadata is
% never superimposed on the visible glyph or silently added to PDF text.
\prop_new:N \g__impe_glyph_logical_prop
\prop_new:N \g__impe_glyph_description_prop
\tl_new:N \l__impe_glyph_logical_tl
\tl_new:N \l__impe_glyph_description_tl
\cs_generate_variant:Nn \prop_gput:Nnn { NnV }
\keys_define:nn { impe/glyph/register }
  {
    logical .tl_set:N = \l__impe_glyph_logical_tl,
    description .tl_set:N = \l__impe_glyph_description_tl,
  }
\NewDocumentCommand \IMPEGlyphRegister { O{} m m }
  {
    \group_begin:
      \tl_clear:N \l__impe_glyph_logical_tl
      \tl_clear:N \l__impe_glyph_description_tl
      \keys_set:nn { impe/glyph/register } {#1}
      \GlyphRegister {#2} {#3}
      \prop_gput:NnV \g__impe_glyph_logical_prop {#2} \l__impe_glyph_logical_tl
      \prop_gput:NnV \g__impe_glyph_description_prop {#2} \l__impe_glyph_description_tl
    \group_end:
  }
\NewDocumentCommand \IMPEGlyphLogical { m }
  {
    \prop_get:NnNT \g__impe_glyph_logical_prop {#1} \l_tmpa_tl
      { \tl_use:N \l_tmpa_tl }
  }
\NewDocumentCommand \IMPEGlyphDescription { m }
  {
    \prop_get:NnNT \g__impe_glyph_description_prop {#1} \l_tmpa_tl
      { \tl_use:N \l_tmpa_tl }
  }

\tl_new:N \l__impe_glyph_scale_tl
\tl_new:N \l__impe_glyph_raise_tl
\tl_new:N \l__impe_glyph_trim_tl
\keys_define:nn { impe/glyph/use }
  {
    scale .tl_set:N = \l__impe_glyph_scale_tl,
    raise .tl_set:N = \l__impe_glyph_raise_tl,
    trim .tl_set:N = \l__impe_glyph_trim_tl,
  }
\cs_new_protected:Npn \__impe_glyph_draw:n #1
  {
    \use:x
      {
        \exp_not:N \Glyph
          [ scale=\tl_use:N \l__impe_glyph_scale_tl,
            raise=\tl_use:N \l__impe_glyph_raise_tl ]
          { \exp_not:n {#1} }
      }
  }
\NewDocumentCommand \IMPEGlyph { O{} m }
  {
    \group_begin:
      \tl_set:Nn \l__impe_glyph_scale_tl {1}
      \tl_set:Nn \l__impe_glyph_raise_tl {0pt}
      \tl_clear:N \l__impe_glyph_trim_tl
      \keys_set:nn { impe/glyph/use } {#1}
      \tl_if_blank:VTF \l__impe_glyph_trim_tl
        { \__impe_glyph_draw:n {#2} }
        {
          % Optical adjustment only: original asset/cache are unchanged.
          \adjustbox{trim={\l__impe_glyph_trim_tl},clip}
            {\__impe_glyph_draw:n {#2}}
        }
    \group_end:
  }
\ExplSyntaxOff
\endinput
