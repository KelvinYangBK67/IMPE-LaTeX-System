\cs_new_protected:Npn \__layout_apply_component:n #1
  {
    \prop_if_in:NnTF \g__layout_component_loaded_prop {#1}
      { }
      {
        \cs_if_exist:cTF { __layout_component_#1: }
          {
            \use:c { __layout_component_#1: }
            \prop_put:Nnn \g__layout_component_loaded_prop {#1} { loaded }
          }
          {
            \PackageError{layout-modules}
              {Unknown~internal~layout~component~`#1`}
              {Define~the~component~before~using~it~from~a~preset.}
          }
      }
  }

\cs_new_protected:Npn \__layout_apply_component_list:n #1
  {
    \tl_if_blank:nTF {#1}
      { }
      { \clist_map_inline:nn {#1} { \__layout_apply_component:n {##1} } }
  }

\newcommand{\NextLayoutTypeDefaults}[4]{%
  \RequirePackage{setspace}%
  \setstretch{1}%
  % Respect explicit 11pt/12pt base-class options: only replace the 10pt
  % class default with the curated IMPE font and baseline dimensions.
  \ifnum\@ptsize=\z@
    \renewcommand{\normalsize}{%
      \@setfontsize\normalsize{#1}{#2}%
      \abovedisplayskip 9pt plus 2pt minus 4pt%
      \belowdisplayskip \abovedisplayskip
      \abovedisplayshortskip 0pt plus 2pt%
      \belowdisplayshortskip 5pt plus 2pt minus 2pt%
    }%
    \renewcommand{\footnotesize}{\@setfontsize\footnotesize{#3}{#4}}%
    \normalsize
  \fi
}
\newcommand{\NextLayoutBookSections}[2]{%
  \renewcommand{\section}{\@startsection{section}{1}{\z@}%
    {-3.2ex \@plus -1ex \@minus -.2ex}{1.8ex \@plus .2ex}%
    {\normalfont\bfseries\fontsize{#1}{17pt}\selectfont}}%
  \renewcommand{\subsection}{\@startsection{subsection}{2}{\z@}%
    {-2.6ex \@plus -.8ex \@minus -.2ex}{1.2ex \@plus .2ex}%
    {\normalfont\bfseries\fontsize{#2}{15pt}\selectfont}}%
}
\newcommand{\NextLayoutPaperSections}[2]{%
  \renewcommand{\section}{\@startsection{section}{1}{\z@}%
    {-3ex \@plus -1ex \@minus -.2ex}{1.4ex \@plus .2ex}%
    {\normalfont\bfseries\fontsize{#1}{18pt}\selectfont}}%
  \renewcommand{\subsection}{\@startsection{subsection}{2}{\z@}%
    {-2.5ex \@plus -.8ex \@minus -.2ex}{1ex \@plus .2ex}%
    {\normalfont\bfseries\fontsize{#2}{16pt}\selectfont}}%
}
\newcommand{\NextLayoutBookChapterHead}{%
  \def\@makechapterhead##1{%
    \vspace*{30pt}%
    {\parindent\z@\raggedright\normalfont
      \ifnum\c@secnumdepth>\m@ne
        {\large\@chapapp\space\thechapter\par\nobreak\vskip 12pt}%
      \fi
      \interlinepenalty\@M
      {\fontsize{18pt}{23pt}\selectfont\bfseries ##1\par\nobreak}%
      \vskip 26pt}}%
  \def\@makeschapterhead##1{%
    \vspace*{30pt}%
    {\parindent\z@\raggedright\normalfont
      \interlinepenalty\@M
      {\fontsize{18pt}{23pt}\selectfont\bfseries ##1\par\nobreak}%
      \vskip 26pt}}%
}

      \fi
      \interlinepenalty\@M
      {\fontsize{18pt}{23pt}\selectfont\bfseries ##1\par\nobreak}%
      \vskip 22pt}}%
  \def\@makeschapterhead##1{%
    \vspace*{22pt}%

      \interlinepenalty\@M
      {\fontsize{18pt}{23pt}\selectfont\bfseries ##1\par\nobreak}%
      \vskip 22pt}}%
}
