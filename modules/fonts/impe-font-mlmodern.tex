% Special font module: mlmodern
%
% MLModern is a traditional LaTeX font package, not an OpenType family.
% Loading it here lets the font registry expose `fonts={mlmodern}` while
% keeping its NFSS/package behavior separate from fontspec registrations.

\ExplSyntaxOn
\tl_new:N \l__font_mlmodern_rmdefault_tl
\tl_new:N \l__font_mlmodern_sfdefault_tl
\tl_new:N \l__font_mlmodern_ttdefault_tl
\tl_new:N \l__font_mlmodern_bfdefault_tl

\cs_new_protected:Npn \__font_define_special_mlmodern:
  {
    % Loading a local preset must not replace the document's Unicode defaults.
    \tl_set:No \l__font_mlmodern_rmdefault_tl {\rmdefault}
    \tl_set:No \l__font_mlmodern_sfdefault_tl {\sfdefault}
    \tl_set:No \l__font_mlmodern_ttdefault_tl {\ttdefault}
    \tl_set:No \l__font_mlmodern_bfdefault_tl {\bfdefault}
    \RequirePackage{mlmodern}
    \cs_set:Npx \rmdefault {\exp_not:V \l__font_mlmodern_rmdefault_tl}
    \cs_set:Npx \sfdefault {\exp_not:V \l__font_mlmodern_sfdefault_tl}
    \cs_set:Npx \ttdefault {\exp_not:V \l__font_mlmodern_ttdefault_tl}
    \cs_set:Npx \bfdefault {\exp_not:V \l__font_mlmodern_bfdefault_tl}
    % The local mapper rebinds sffamily/ttfamily to these callbacks. Calling
    % those switches here recurses forever. Select the Type1 NFSS faces directly.
    \cs_set_protected:cpn { __fontrm_\l__font_name_tl }
      { \fontencoding{T1}\fontfamily{mlmr}\selectfont }
    \cs_set_protected:cpn { __fontsf_\l__font_name_tl }
      { \fontencoding{T1}\fontfamily{mlmss}\selectfont }
    \cs_set_protected:cpn { __fonttt_\l__font_name_tl }
      { \fontencoding{T1}\fontfamily{mlmtt}\selectfont }
    \exp_args:NV \__font_define_behavior_hooks:n \l__font_name_tl
    \exp_args:NVV \__font_define_public_command:nn
      \l__font_command_tl
      \l__font_name_tl
  }

\ExplSyntaxOff
