% Khitan Small Script: the public local command uses XSR native stacking.
% Unicode cluster separators and U+16FE4 fillers are preserved unchanged.
\RequirePackage{xsr-detector-active}
\RequirePackage{xsr-khitan}
\NewDocumentCommand \IMPEKhitanText { m } { \xsrKhitanText {#1} }
\ExplSyntaxOn

\tl_new:N \l__font_khs_content_tl

% The normal eight-style specimen wrappers are formatting requests, not
% Khitan source characters. Unwrap them before passing Unicode to XSR.
% XSR currently uses the selected glyph font; unavailable style faces do not
% alter the cluster geometry.
\cs_new_protected:Npn \__font_khs_prepare_content:n #1
  {
    \tl_set:Nn \l__font_khs_content_tl {#1}
    \__font_khs_strip_wrappers:
  }

\cs_new_protected:Npn \__font_khs_strip_wrappers:
  {
    \tl_if_head_eq_meaning:VNTF \l__font_khs_content_tl \textsf
      { \__font_khs_unwrap_textsf: }
      {
        \tl_if_head_eq_meaning:VNTF \l__font_khs_content_tl \texttt
          { \__font_khs_unwrap_texttt: }
          {
            \tl_if_head_eq_meaning:VNTF \l__font_khs_content_tl \textbf
              { \__font_khs_unwrap_textbf: }
              {
                \tl_if_head_eq_meaning:VNTF \l__font_khs_content_tl \textit
                  { \__font_khs_unwrap_textit: }
                  { }
              }
          }
      }
  }

\cs_new_protected:Npn \__font_khs_unwrap_textsf:
  { \exp_after:wN \__font_khs_unwrap_textsf_aux:w \l__font_khs_content_tl \q_stop }
\cs_new_protected:Npn \__font_khs_unwrap_textsf_aux:w \textsf #1#2 \q_stop
  {
    \tl_set:Nn \l__font_khs_content_tl {#1}
    \__font_khs_strip_wrappers:
  }

\cs_new_protected:Npn \__font_khs_unwrap_texttt:
  { \exp_after:wN \__font_khs_unwrap_texttt_aux:w \l__font_khs_content_tl \q_stop }
\cs_new_protected:Npn \__font_khs_unwrap_texttt_aux:w \texttt #1#2 \q_stop
  {
    \tl_set:Nn \l__font_khs_content_tl {#1}
    \__font_khs_strip_wrappers:
  }

\cs_new_protected:Npn \__font_khs_unwrap_textbf:
  { \exp_after:wN \__font_khs_unwrap_textbf_aux:w \l__font_khs_content_tl \q_stop }
\cs_new_protected:Npn \__font_khs_unwrap_textbf_aux:w \textbf #1#2 \q_stop
  {
    \tl_set:Nn \l__font_khs_content_tl {#1}
    \__font_khs_strip_wrappers:
  }

\cs_new_protected:Npn \__font_khs_unwrap_textit:
  { \exp_after:wN \__font_khs_unwrap_textit_aux:w \l__font_khs_content_tl \q_stop }
\cs_new_protected:Npn \__font_khs_unwrap_textit_aux:w \textit #1#2 \q_stop
  {
    \tl_set:Nn \l__font_khs_content_tl {#1}
    \__font_khs_strip_wrappers:
  }
\cs_new_protected:Npn \__font_khs_render_xsr:n #1
  {
    \__font_khs_prepare_content:n {#1}
    \exp_args:NV \xsrKhitanText \l__font_khs_content_tl
  }

\cs_new_protected:Npn \__font_define_public_command_khitan:nn #1#2
  {
    % KHS is the canonical stacked entry point, including through \Font.
    \cs_set_protected:cpn {#1} ##1
      {
        {
          \bool_set_true:N \l__font_local_override_bool
          \use:c { __font_prepare_local_#2: }
          \__font_khs_render_xsr:n {##1}
        }
      }
    % Existing explicit stack names remain convenient aliases.
    \exp_args:Nc \DeclareDocumentCommand {#1stack} {m}
      { \__font_khs_render_xsr:n {##1} }
    \exp_args:Nc \DeclareDocumentCommand {#1stackblock} {+m}
      {
        \par \noindent
        \__font_khs_render_xsr:n {##1}
        \par
      }
  }

\cs_new_protected:Npn \__font_define_special_khitan_small:
  {
    \tl_clear:N \l__font_script_tl
    \tl_clear:N \l__font_language_tl
    \tl_set:Nn \l__font_backend_tl { generic }
    \__font_special_use_generic:
    \exp_args:NV \xsrKhitanDefaultFont \l__font_regular_tl
    \exp_args:NVV \__font_define_public_command_khitan:nn
      \l__font_command_tl
      \l__font_name_tl
  }
\ExplSyntaxOff
