% Khitan Small Script font route: linear text belongs to IMPE, stacked
% clusters belong to XSR.  No second stacking algorithm is maintained here.
\ExplSyntaxOn
\cs_new_protected:Npn \__font_khs_stack:n #1
  {
    \cs_if_exist:NTF \xsrKhitanText
      { \xsrKhitanText{#1} }
      {
        \PackageError{impe-khitan}
          {XSR integration is required for stacked Khitan Small Script}
          {Load \string\UseFeature{xsr} in the preamble, and select an XSR
           Khitan font file with \string\xsrKhitanDefaultFont{path}.}
      }
  }
\cs_new_protected:Npn \__font_define_public_command_khitan:nn #1#2
  {
    \exp_args:Nc \DeclareDocumentCommand {#1stack} {m}
      { \__font_khs_stack:n {##1} }
    \exp_args:Nc \DeclareDocumentCommand {#1stackblock} {+m}
      {
        \par \noindent
        \__font_khs_stack:n {##1}
        \par
      }
  }
\cs_new_protected:Npn \__font_define_special_khitan_small:
  {
    \tl_clear:N \l__font_script_tl
    \tl_clear:N \l__font_language_tl
    \tl_set:Nn \l__font_backend_tl { generic }
    \__font_special_use_generic:
    % The ordinary \KHS remains the local linear font route.
    \exp_args:NVV \__font_define_public_command_khitan:nn
      \l__font_command_tl
      \l__font_name_tl
  }
\ExplSyntaxOff
