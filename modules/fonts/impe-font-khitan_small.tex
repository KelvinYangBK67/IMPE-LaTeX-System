% Khitan Small Script special builder.
%
% The current TeX/fontspec shaping path does not reliably expose the font's
% native cluster substitution behaviour, so IMPE LaTeX System composes Khitan Small Script
% clusters explicitly from the encoded character stream.
%
% Model used here:
% - clusters are separated by spaces
% - Type A starts with two adjacent characters
% - Type B inserts U+16FE4 KHITAN SMALL SCRIPT FILLER after the first character
% - Type A lays out rows of pairs, with a final centred character if odd
% - Type B lays out a first centred character, then applies the Type A rule to
%   the remaining characters

\RequirePackage{xsr-detector-active}
\RequirePackage{xsr-khitan}
\NewDocumentCommand \IMPEKhitanText { m } { \xsrKhitanText {#1} }
\ExplSyntaxOn

\tl_const:Nn \c__font_khs_filler_tl { 𖿤 }

\cs_new_protected:Npn \__font_khs_reset_style_parse:
  {
    \tl_clear:N \l__font_khs_style_family_tl
    \tl_clear:N \l__font_khs_style_series_tl
    \tl_clear:N \l__font_khs_style_shape_tl
    \tl_clear:N \l__font_khs_style_setup_tl
  }

\cs_new_protected:Npn \__font_khs_apply_style_setup:
  {
    \tl_clear:N \l__font_khs_style_setup_tl
    \tl_if_blank:VF \l__font_khs_style_family_tl
      { \tl_put_right:NV \l__font_khs_style_setup_tl \l__font_khs_style_family_tl }
    \tl_if_blank:VF \l__font_khs_style_series_tl
      { \tl_put_right:NV \l__font_khs_style_setup_tl \l__font_khs_style_series_tl }
    \tl_if_blank:VF \l__font_khs_style_shape_tl
      { \tl_put_right:NV \l__font_khs_style_setup_tl \l__font_khs_style_shape_tl }
  }

\cs_new_protected:Npn \__font_khs_prepare_content:n #1
  {
    \tl_set:Nn \l__font_khs_content_tl {#1}
    \__font_khs_reset_style_parse:
    \__font_khs_strip_wrappers:
    \__font_khs_apply_style_setup:
  }

\cs_new_protected:Npn \__font_khs_strip_wrappers:
  {
    \tl_if_head_eq_meaning:VNTF \l__font_khs_content_tl \textsf
      { \__font_khs_unwrap_textsf: }
      {
        \tl_if_head_eq_meaning:VNTF \l__font_khs_content_tl \texttt
          { \__font_khs_unwrap_texttt: }
          {
            \tl_if_head_eq_meaning:VNTF \l__font_khs_content_tl \textbf
              { \__font_khs_unwrap_textbf: }
              {
                \tl_if_head_eq_meaning:VNTF \l__font_khs_content_tl \textit
                  { \__font_khs_unwrap_textit: }
                  { }
              }
          }
      }
  }

\cs_new_protected:Npn \__font_khs_unwrap_textsf:
  { \exp_after:wN \__font_khs_unwrap_textsf_aux:w \l__font_khs_content_tl \q_stop }
\cs_new_protected:Npn \__font_khs_unwrap_textsf_aux:w \textsf #1#2 \q_stop
  {
    \tl_set:Nn \l__font_khs_style_family_tl { \sffamily }
    \tl_set:Nn \l__font_khs_content_tl {#1}
    \__font_khs_strip_wrappers:
  }

\cs_new_protected:Npn \__font_khs_unwrap_texttt:

  {
    \tl_set:Nn \l__font_khs_style_family_tl { \ttfamily }
    \tl_set:Nn \l__font_khs_content_tl {#1}
    \__font_khs_strip_wrappers:
  }

\cs_new_protected:Npn \__font_khs_unwrap_textbf:
  { \exp_after:wN \__font_khs_unwrap_textbf_aux:w \l__font_khs_content_tl \q_stop }
\cs_new_protected:Npn \__font_khs_unwrap_textbf_aux:w \textbf #1#2 \q_stop
  {
    \tl_set:Nn \l__font_khs_style_series_tl { \bfseries }
    \tl_set:Nn \l__font_khs_content_tl {#1}

\cs_new_protected:Npn \__font_khs_unwrap_textit:
  { \exp_after:wN \__font_khs_unwrap_textit_aux:w \l__font_khs_content_tl \q_stop }
\cs_new_protected:Npn \__font_khs_unwrap_textit_aux:w \textit #1#2 \q_stop
  {
    \tl_set:Nn \l__font_khs_style_shape_tl { \itshape }
    \tl_set:Nn \l__font_khs_content_tl {#1}
    \__font_khs_strip_wrappers:
  }
\cs_new:Npn \__font_khs_strip_fillers:n #1
  {
    \tl_clear:N \l_tmpa_tl
    \tl_map_inline:nn {#1}
      {
        \str_if_eq:eeF
          { \tl_to_str:n {##1} }
          { \tl_to_str:V \c__font_khs_filler_tl }
          { \tl_put_right:Nn \l_tmpa_tl {##1} }
      }
    \tl_use:N \l_tmpa_tl
  }

\cs_new_protected:Npn \__font_khs_render_linear:n #1
  {
    \__font_khs_prepare_content:n {#1}
    % Active Unicode glyph tokens must become ordinary characters in linear mode.
    \tl_set:Nx \l_tmpa_tl { \tl_to_str:V \l__font_khs_content_tl }
    \tl_set:Nx \l_tmpb_tl { \tl_to_str:V \c__font_khs_filler_tl }
    \tl_replace_all:NVn \l_tmpa_tl \l_tmpb_tl { }
    \cs_if_exist:NT \XeTeXinterchartokenstate
      { \XeTeXinterchartokenstate = 0 }
    \tl_use:N \l__font_khs_style_setup_tl
    \tl_use:N \l_tmpa_tl
  }
\cs_new_protected:Npn \__font_khs_stack:n #1
  {
    \xsrKhitanText{#1}
  }
\cs_new_protected:Npn \__font_define_public_command_khitan:nn #1#2
  {
    \exp_args:Nc \DeclareDocumentCommand {#1stack} {m}
      { \__font_khs_stack:n {##1} }
    \exp_args:Nc \DeclareDocumentCommand {#1stackblock} {+m}
      {
        \par \noindent
        \__font_khs_stack:n {##1}
        \par
      }
  }
\cs_new_protected:Npn \__font_define_public_command_khitan_linear:nn #1#2
  {
    \cs_set_protected:cpn { #1 } ##1
      {
        {
          \use:c { __font_prepare_local_#2: }
          \__font_khs_render_linear:n {##1}
        }
      }
  }

    \exp_args:NVV \__font_define_public_command_khitan_linear:nn
      \l__font_command_tl
      \l__font_name_tl
    % Stacked layout is delegated to XSR; no IMPE stack geometry remains.

    \exp_args:NVV \__font_define_public_command_khitan:nn
      \l__font_command_tl

  }
