% Parameter overlay for current-document layouts.
% Preset components remain the authority for unspecified values.
\RequirePackage{expl3}
\RequirePackage{xparse}
\makeatletter
\ExplSyntaxOn
\prop_new:N \g__impe_layout_config_seen_prop
\prop_new:N \l__impe_layout_scope_seen_prop
\bool_new:N \l__impe_layout_scope_bool
\bool_new:N \g__impe_layout_config_used_bool

\cs_new_protected:Npn \__impe_layout_config_apply:nn #1#2
  {
    \str_case:nnF {#1}
      {
        {page.top}{\__impe_layout_geometry:n {top={#2}}}
        {page.bottom}{\__impe_layout_geometry:n {bottom={#2}}}
        {page.inner}{\__impe_layout_geometry:n {inner={#2}}}
        {page.outer}{\__impe_layout_geometry:n {outer={#2}}}
        {page.left}{\__impe_layout_geometry:n {left={#2}}}
        {page.right}{\__impe_layout_geometry:n {right={#2}}}
        {page.binding-offset}{\__impe_layout_geometry:n {bindingoffset={#2}}}
        {text.line-stretch}{\setstretch{#2}}
        {text.par-indent}{\setlength{\parindent}{#2}}
        {text.par-skip}{\setlength{\parskip}{#2}}
        {text.left-skip}{\setlength{\leftskip}{#2}}
        {text.right-skip}{\setlength{\rightskip}{#2}}
        {text.list-line-stretch}{\def\IMPEListLineStretch{#2}}
        {book.chapter-opening}
          {
            \str_case:nnF {#2}
              {
                {right}{\cs_if_exist:NT \@openrighttrue {\@openrighttrue}}
                {any}{\cs_if_exist:NT \@openrightfalse {\@openrightfalse}}
              }
              {\PackageError{impe-layout}{Invalid book.chapter-opening: #2}{Use right or any.}}
          }
        {headers.style}
          {
            \UseFeature{headers}
            \HeaderStyle{#2}
          }
      }
      {\PackageError{impe-layout}{Unknown parameter #1}{Check the documented layout keys.}}
  }

\cs_new_protected:Npn \__impe_layout_geometry:n #1
  {
    \@ifpackageloaded{geometry}
      {\geometry{#1}}
      {\RequirePackage{geometry}\geometry{#1}}
  }

\cs_new_protected:Npn \__impe_layout_config_assign:nnn #1#2#3
  {
    \tl_if_blank:nF {#3}
      {
        \bool_if:NTF \l__impe_layout_scope_bool
          {
            \str_if_eq:nnTF {#2}{local}
              {
                \prop_if_in:NnTF \l__impe_layout_scope_seen_prop {#1}
                  {\PackageError{impe-layout}{Duplicate local parameter #1}{Set each parameter once per scope.}}
                  {
                    \prop_put:Nnn \l__impe_layout_scope_seen_prop {#1}{set}
                    \__impe_layout_config_apply:nn {#1}{#3}
                  }
              }
              {\PackageError{impe-layout}{Page or feature parameter #1 is not local}{Use LayoutSetup in the preamble.}}
          }
          {
            \prop_if_in:NnTF \g__impe_layout_config_seen_prop {#1}
              {\PackageError{impe-layout}{Duplicate document parameter #1}{Set each parameter once per document.}}
              {
                \prop_gput:Nnn \g__impe_layout_config_seen_prop {#1}{set}
                \bool_gset_true:N \g__impe_layout_config_used_bool
                \__impe_layout_config_apply:nn {#1}{#3}
              }
          }
      }
  }

\keys_define:nn {impe/layout-config}
  {
    page.top .code:n = {\__impe_layout_config_assign:nnn {page.top}{global}{#1}},
    page.bottom .code:n = {\__impe_layout_config_assign:nnn {page.bottom}{global}{#1}},
    page.inner .code:n = {\__impe_layout_config_assign:nnn {page.inner}{global}{#1}},
    page.outer .code:n = {\__impe_layout_config_assign:nnn {page.outer}{global}{#1}},
    page.left .code:n = {\__impe_layout_config_assign:nnn {page.left}{global}{#1}},
    page.right .code:n = {\__impe_layout_config_assign:nnn {page.right}{global}{#1}},
    page.binding-offset .code:n = {\__impe_layout_config_assign:nnn {page.binding-offset}{global}{#1}},
    text.line-stretch .code:n = {\__impe_layout_config_assign:nnn {text.line-stretch}{local}{#1}},
    text.par-indent .code:n = {\__impe_layout_config_assign:nnn {text.par-indent}{local}{#1}},
    text.par-skip .code:n = {\__impe_layout_config_assign:nnn {text.par-skip}{local}{#1}},
    text.left-skip .code:n = {\__impe_layout_config_assign:nnn {text.left-skip}{local}{#1}},
    text.right-skip .code:n = {\__impe_layout_config_assign:nnn {text.right-skip}{local}{#1}},
    text.list-line-stretch .code:n = {\__impe_layout_config_assign:nnn {text.list-line-stretch}{local}{#1}},
    book.chapter-opening .code:n = {\__impe_layout_config_assign:nnn {book.chapter-opening}{global}{#1}},
    headers.style .code:n = {\__impe_layout_config_assign:nnn {headers.style}{global}{#1}}
  }

% Document-wide settings intentionally are preamble-only. Multiple calls may
% assign different keys, but an already assigned key must never be assigned again.
\NewDocumentCommand{\LayoutSetup}{m}
  {\keys_set:nn {impe/layout-config}{#1}}
\NewDocumentCommand{\LayoutSet}{mm}
  {\keys_set:nn {impe/layout-config}{#1={#2}}}
\@onlypreamble\LayoutSetup
\@onlypreamble\LayoutSet

% Paragraph completion must precede scope restoration: \par inside the group
% ensures the local baseline and paragraph parameters remain active.
\NewDocumentEnvironment{LayoutScope}{m}
  {
    \par
    \group_begin:
    \bool_set_true:N \l__impe_layout_scope_bool
    \prop_clear:N \l__impe_layout_scope_seen_prop
    \keys_set:nn {impe/layout-config}{#1}
    \ignorespaces
  }
  {
    \par
    \group_end:
    \ignorespacesafterend
  }
\ExplSyntaxOff
\makeatother
