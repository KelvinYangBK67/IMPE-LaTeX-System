\cs_new_protected:Npn \__writing_fallback_candidate:n #1
  {
    % xeCJK can only manage CJK-shaped fallback families. Other scripts
    % are still tracked by the global pool for their own renderers.
    \prop_get:NnN \g__writing_pool_profile_prop {#1}
      \l__writing_fallback_candidate_profile_tl
    \bool_set_false:N \l_tmpa_bool
    \str_case:Vn \l__writing_fallback_candidate_profile_tl
      {
        {han}{\bool_set_true:N \l_tmpa_bool}
        {chinese}{\bool_set_true:N \l_tmpa_bool}
        {japanese}{\bool_set_true:N \l_tmpa_bool}
        {korean}{\bool_set_true:N \l_tmpa_bool}
        {tangut}{\bool_set_true:N \l_tmpa_bool}
      }
    \bool_if:NT \l_tmpa_bool
      {
    \prop_get:NnN \g__writing_pool_font_prop {#1}
      \l__writing_fallback_candidate_font_tl
    \prop_get:NnN \g__writing_pool_path_prop {#1}
      \l__writing_fallback_candidate_path_tl
    \tl_if_blank:VF \l__writing_fallback_candidate_font_tl
      {
        % Once the core loader resolves a bundled face, regular contains an
        % absolute filename and path is empty. xeCJK cannot consume that
        % absolute filename as a family name (TU//home/... NFSS failures).
        % Split into the officially supported [Path=directory]{basename}.
        \tl_if_blank:VTF \l__writing_fallback_candidate_path_tl
          {
            \tl_if_in:VnTF \l__writing_fallback_candidate_font_tl {/}
              {
                \tl_set:NV \l_tmpa_tl \l__writing_fallback_candidate_font_tl
                \tl_set:NV \l_tmpb_tl \l__writing_fallback_candidate_font_tl
                \regex_replace_once:nnN { [^/]+ \Z } {} \l_tmpa_tl
                \regex_replace_once:nnN { \A .*/ } {} \l_tmpb_tl
                \tl_set:Nx \l_tmpc_tl
                  {
                    [Path={\exp_not:V \l_tmpa_tl}]
                    {\exp_not:V \l_tmpb_tl}
                  }
                \clist_put_right:NV \l__writing_fallback_candidates_clist
                  \l_tmpc_tl
              }
              {
                \clist_put_right:NV \l__writing_fallback_candidates_clist
                  \l__writing_fallback_candidate_font_tl
              }
          }
          {
            \tl_set:Nx \l_tmpa_tl
              {
                [Path={\exp_not:V \l__writing_fallback_candidate_path_tl}]
                {\exp_not:V \l__writing_fallback_candidate_font_tl}
              }
            \clist_put_right:NV \l__writing_fallback_candidates_clist \l_tmpa_tl
          }
      }
      }
  }

% Add one explicitly declared face, and force its use for the requested NFSS
% style. Passing the same face as Bold/Italic/BI is essential: otherwise
% fontspec may substitute a different glyph-bearing face behind the matrix.
\cs_new_protected:Npn \__writing_fallback_add_face:nn #1#2
  {
    \tl_set:Nx \l__writing_fallback_face_key_tl {#1/#2}
    \prop_get:NVNT \g__writing_face_font_prop
      \l__writing_fallback_face_key_tl
      \l__writing_fallback_candidate_font_tl
      {
        \prop_get:NVN \g__writing_face_path_prop
          \l__writing_fallback_face_key_tl
          \l__writing_fallback_candidate_path_tl
        \str_if_eq:VnT \l__writing_fallback_candidate_path_tl
          {__impe_none__}
          { \tl_clear:N \l__writing_fallback_candidate_path_tl }
        \tl_if_blank:VT \l__writing_fallback_candidate_path_tl
          {
            \tl_if_in:VnT \l__writing_fallback_candidate_font_tl {/}
              {
                \tl_set:NV \l__writing_fallback_candidate_path_tl
                  \l__writing_fallback_candidate_font_tl
                \regex_replace_once:nnN { [^/]+ \Z } {}
                  \l__writing_fallback_candidate_path_tl
                \regex_replace_once:nnN { \A .*/ } {}
                  \l__writing_fallback_candidate_font_tl
              }
          }
        \tl_set:Nx \l__writing_fallback_font_key_tl
          {
            \tl_to_str:V \l__writing_fallback_candidate_path_tl /
            \tl_to_str:V \l__writing_fallback_candidate_font_tl
          }
        \prop_if_in:NVF \l__writing_fallback_seen_prop
          \l__writing_fallback_font_key_tl
          {
            \prop_put:NVn \l__writing_fallback_seen_prop
              \l__writing_fallback_font_key_tl {seen}
            \tl_clear:N \l__writing_fallback_face_options_tl
            \tl_if_blank:VF \l__writing_fallback_candidate_path_tl
              {
                \tl_put_right:Nx \l__writing_fallback_face_options_tl
                  {Path={\exp_not:V \l__writing_fallback_candidate_path_tl},}
              }
            \int_case:nn \l__writing_fallback_style_int
              {
                {1}{
                  \tl_put_right:Nx \l__writing_fallback_face_options_tl
                    {ItalicFont={\exp_not:V \l__writing_fallback_candidate_font_tl},}
                }
                {2}{
                  \tl_put_right:Nx \l__writing_fallback_face_options_tl
                    {BoldFont={\exp_not:V \l__writing_fallback_candidate_font_tl},}
                }
                {3}{
                  \tl_put_right:Nx \l__writing_fallback_face_options_tl
                    {BoldItalicFont={\exp_not:V \l__writing_fallback_candidate_font_tl},}
                }
              }
            \tl_if_blank:VTF \l__writing_fallback_face_options_tl
              {
                \clist_put_right:NV \l__writing_fallback_candidates_clist
                  \l__writing_fallback_candidate_font_tl
              }
              {
                \regex_replace_once:nnN { , \Z } {}
                  \l__writing_fallback_face_options_tl
                % Extra outer group protects commas within fontspec options
                % from xeCJK's clist of fallback candidates.
                \tl_set:Nx \l__writing_fallback_face_entry_tl
                  {
                    {
                      [\exp_not:V \l__writing_fallback_face_options_tl]
                      {\exp_not:V \l__writing_fallback_candidate_font_tl}
                    }
                  }
                \clist_put_right:NV \l__writing_fallback_candidates_clist
                  \l__writing_fallback_face_entry_tl
              }
          }
      }
  }

            % ##2 is xeCJK's registered family (rm/sf/private family).
            % ##1 is its current NFSS font and is not a registry key.
            \__writing_fallback_refresh_family:n {##2}
            \__writing_base_xecjk_fallback_loop:nnNN
              {##1}{##2}##3##4
          }
      }
    \cs_if_exist:NT \__xeCJK_select_fallback_font:Nnnn
      {
        \cs_new_eq:NN \__writing_base_xecjk_select_fallback_font:Nnnn
          \__xeCJK_select_fallback_font:Nnnn
        \cs_set_protected:Npn \__xeCJK_select_fallback_font:Nnnn ##1##2##3##4
          {
            % xeCJK's selector receives its own resolved NFSS family,
            % not necessarily our registered Writing family name. Track
            % a global chain generation per cached font control instead.
            \tl_set:Nx \l_tmpb_tl {\cs_to_str:N ##1}
            \prop_get:NVNT \g__writing_fallback_cache_revision_prop
              \l_tmpb_tl \l_tmpc_tl
              {
                \int_compare:nNnF {\l_tmpc_tl} =
                  {\g__writing_fallback_revision_int}
                  { \cs_undefine:N ##1 }
              }
            \prop_gput:NVx \g__writing_fallback_cache_revision_prop
              \l_tmpb_tl {\int_use:N \g__writing_fallback_revision_int}
            \__writing_base_xecjk_select_fallback_font:Nnnn
              ##1{##2}{##3}{##4}
          }
      }
    \cs_if_exist:NT \__xeCJK_fallback_loop_aux:nnnNN
      {
        \cs_new_eq:NN
          \__writing_base_xecjk_fallback_loop_aux:nnnNN
          \__xeCJK_fallback_loop_aux:nnnNN
        \cs_set_protected:Npn \__xeCJK_fallback_loop_aux:nnnNN ##1##2##3##4##5
          {
            \xeCJK_glyph_if_exist:NT ##5
              { \__writing_fallback_report:nnn {##2} {##1} {##5} }
            \__writing_base_xecjk_fallback_loop_aux:nnnNN
              {##1}{##2}{##3}##4##5
          }
      }
