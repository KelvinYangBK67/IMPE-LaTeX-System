% Parse [optional pack, key=value, ...]profile.  TeX brace groups are
% restored when mapping; commas inside braced file names stay protected.
\cs_new_protected:Npn \__writing_entry_char:n #1
  {
    \str_if_eq:eeTF {\tl_to_str:n {#1}} {[}
      { \bool_set_true:N \l__writing_in_options_bool }
      {
        \str_if_eq:eeTF {\tl_to_str:n {#1}} {]}
          { \bool_set_false:N \l__writing_in_options_bool }
          {
            \bool_if:NTF \l__writing_in_options_bool
              { \tl_put_right:Nn \l__writing_options_tl {#1} }
              { \tl_put_right:Nn \l__writing_profile_tl {#1} }
          }
      }
  }
\cs_new_protected:Npn \__writing_parse_entry:n #1
  {
    \tl_clear:N \l__writing_options_tl
    \tl_clear:N \l__writing_profile_tl
    \bool_set_false:N \l__writing_in_options_bool
    \tl_map_inline:nn {#1}
      {
        \tl_if_single_token:nTF {##1}
          { \__writing_entry_char:n {##1} }
          {
            \bool_if:NTF \l__writing_in_options_bool
              { \tl_put_right:Nn \l__writing_options_tl { {##1} } }
              { \tl_put_right:Nn \l__writing_profile_tl { {##1} } }
          }
      }
    \tl_replace_all:Nnn \l__writing_profile_tl {\__writing_space_marker:} { ~ }
    \tl_replace_all:Nnn \l__writing_options_tl {\__writing_space_marker:} { ~ }
    \tl_trim_spaces:N \l__writing_profile_tl
    \tl_trim_spaces:N \l__writing_options_tl
    % A bare key=value list is a generic font declaration with no profile.
    \tl_if_blank:VTF \l__writing_options_tl
      {
        \tl_if_in:VnT \l__writing_profile_tl {=}
          {
            \tl_set:NV \l__writing_options_tl \l__writing_profile_tl
            \tl_clear:N \l__writing_profile_tl
          }
      }
      { }
    \bool_lazy_or:nnT
      { ! \tl_if_blank_p:V \l__writing_profile_tl }
      { ! \tl_if_blank_p:V \l__writing_options_tl }
      {
        \bool_if:NTF \l__writing_local_call_bool
          {
            \int_incr:N \l__writing_local_entry_count_int
            \int_compare:nNnTF {\l__writing_local_entry_count_int} = {1}
              { \__writing_apply: }
              {
                \PackageError{impe-writing}{Local~writing~requires~one~binding}
                  {Local~font~calls~cannot~route~multiple~Unicode~profiles.}
              }
          }
          { \__writing_apply: }
      }
  }

            \__writing_error:nn {Writing~profile~has~no~catalog~declaration}

      }
    \bool_if:NT \l__writing_valid_bool
      {
        \__font_reset_state:
        \exp_args:NnV \keys_set:nn {font/declare} \l_tmpa_tl
        \tl_set:NV \l__writing_profile_command_tl \l__font_command_tl
        \tl_set:Nx \l__writing_profile_behavior_tl

            script={\exp_not:V \l__font_script_tl},
            language={\exp_not:V \l__font_language_tl},
            features={\exp_not:V \l__font_features_tl},

\cs_new_protected:Npn \__writing_apply:
  {
    \__writing_read_options:
    \bool_if:NT \l__writing_valid_bool
      {
        \tl_if_blank:VTF \l__writing_profile_tl
          {
            % A pure font has no Unicode routing nor special renderer.
            \tl_set:Nn \l__writing_profile_behavior_tl
              {scriptclass=latin,inlinebehavior=normal,layout={},specialmodule={},script={},language={},features={}}
            \tl_set:Nn \l__writing_selected_scope_tl {local}
            \bool_if:NF \l__writing_local_call_bool
              {
                \__writing_error:nn {Profile-less~fonts~require~local~use}
                  {Use~\string\Font~or~\string\Writing,~or~declare~a~Unicode~profile~for~global~routing.}
              }
          }
          {
            \bool_if:NF \l__writing_local_call_bool
              { \__writing_resolve_unicode_coverage: }
            \__writing_prepare_profile_behavior:
          }
      }
    \bool_if:NT \l__writing_local_call_bool
      { \tl_set:Nn \l__writing_selected_scope_tl {local} }
    \cs_if_exist:NT \l__writing_force_global_bool
      {
        \bool_if:NT \l__writing_force_global_bool
          { \tl_set:Nn \l__writing_selected_scope_tl {global} }
      }
    \bool_if:NT \l__writing_valid_bool
      {
        % Prefer the profile's required scope; use the pack's alternate
        % declaration only if it has no matching local/global form.
        \str_if_eq:VnTF \l__writing_selected_scope_tl {global}
          {
            \prop_get:NVNTF \g__font_family_global_prop \l__writing_source_tl \l_tmpa_tl
              { }
              {
                \prop_get:NVNTF \g__font_family_local_prop \l__writing_source_tl \l_tmpa_tl
                  { }
                  { \__writing_error:nn {Missing~source~declaration} {Check~the~font~catalog.} }
              }
          }
          {
            \prop_get:NVNTF \g__font_family_local_prop \l__writing_source_tl \l_tmpa_tl
              { }
              {
                \prop_get:NVNTF \g__font_family_global_prop \l__writing_source_tl \l_tmpa_tl
                  { }
                  { \__writing_error:nn {Missing~source~declaration} {Check~the~font~catalog.} }
              }
          }
        \bool_if:NT \l__writing_valid_bool
          {
            % Global-only legacy families retain their global CJK/main effects;
            % local families retain their rendering hooks and vertical builders.
            \tl_set:NV \l__writing_saved_fallback_tl \l__font_fallback_mode_tl
            \__font_reset_state:
            \tl_set:Nn \l__font_fallback_mode_tl {soft}
            \exp_args:NnV \keys_set:nn {font/declare} \l_tmpa_tl
            % Do not let the selected font pack override writing behavior.
            \exp_args:NnV \keys_set:nn {font/declare} \l__writing_profile_behavior_tl
            \tl_set:NV \l__font_scope_tl \l__writing_selected_scope_tl
            \tl_set:NV \l__font_decl_family_id_tl \l__writing_profile_source_tl
            \clist_clear:N \l__font_unicode_blocks_clist
            \tl_set:NV \l__font_command_tl \l__writing_profile_command_tl
            % Each binding receives fresh NFSS identifiers: fontspec's
            % \newfontfamily cannot redeclare the same internal control name.
            \int_gincr:N \g__writing_declaration_int
            \tl_set:Nx \l__font_name_tl
              {writing_\l__writing_profile_tl _\int_use:N \g__writing_declaration_int}
            \tl_set:NV \l__writing_public_command_tl \l__font_command_tl
            \tl_if_blank:VTF \l__font_command_tl
              {
                \tl_clear:N \l__writing_public_command_tl
                \tl_set:Nx \l__font_command_tl
                  {impewriting\int_use:N \g__writing_declaration_int}
              }
              {
                % Keep an already registered *legacy* public command intact.
                % A command created by an earlier UseWriting is ours to update.
                \exp_args:NV \cs_if_exist:cT \l__font_command_tl
                  {
                    \prop_if_in:NVF \g__writing_owned_public_command_prop
                      \l__font_command_tl
                      {
                        \tl_clear:N \l__writing_public_command_tl
                        \tl_set:Nx \l__font_command_tl
                          {impewriting\int_use:N \g__writing_declaration_int}
                      }
                  }
              }
            % The private local entry must never replace \SA, \MO, etc.
            \bool_if:NT \l__writing_local_call_bool
              {
                \tl_clear:N \l__writing_public_command_tl
                \tl_set:Nx \l__font_command_tl
                  {impewriting\int_use:N \g__writing_declaration_int}
              }
            \prop_get:NVNT \g__writing_pack_regular_override_prop \l__writing_pack_tl \l_tmpa_tl
              { \tl_set:NV \l__font_regular_tl \l_tmpa_tl }
            \prop_get:NVNT \g__writing_pack_bold_override_prop \l__writing_pack_tl \l_tmpa_tl
              { \tl_set:NV \l__font_bold_tl \l_tmpa_tl }
            \str_if_eq:VnT \l__writing_stylemap_tl {native}
              { \__writing_clear_non_native: }
            \tl_if_blank:VF \l__writing_font_tl
              {
                \tl_set:NV \l__font_regular_tl \l__writing_font_tl
                \tl_clear:N \l__font_path_tl
                \tl_clear:N \l__font_bold_tl
                \tl_clear:N \l__font_italic_tl
                \tl_clear:N \l__font_bolditalic_tl
                \tl_clear:N \l__font_sans_tl
                \tl_clear:N \l__font_sans_bold_tl
                \tl_clear:N \l__font_sans_italic_tl
                \tl_clear:N \l__font_sans_bolditalic_tl
                \tl_clear:N \l__font_mono_tl
                \tl_clear:N \l__font_mono_bold_tl
              }
            \tl_if_blank:VF \l__writing_file_tl
              {
                \tl_set:NV \l__font_regular_tl \l__writing_file_tl
                \tl_clear:N \l__font_path_tl
                \tl_clear:N \l__font_bold_tl
                \tl_clear:N \l__font_italic_tl
                \tl_clear:N \l__font_bolditalic_tl
                \tl_clear:N \l__font_sans_tl
                \tl_clear:N \l__font_sans_bold_tl
                \tl_clear:N \l__font_sans_italic_tl
                \tl_clear:N \l__font_sans_bolditalic_tl
                \tl_clear:N \l__font_mono_tl
                \tl_clear:N \l__font_mono_bold_tl
              }
            \tl_if_blank:VF \l__writing_writing_tl
              {
                \str_if_eq:VnTF \l__writing_writing_tl {horizontal}
                  { \tl_clear:N \l__font_layout_tl }
                  { \tl_set:Nn \l__font_layout_tl {vertical} }
              }
            \bool_if:NT \l__writing_local_call_bool
              {
                \tl_if_blank:VT \l__writing_writing_tl
                  { \tl_clear:N \l__font_layout_tl }
                \tl_clear:N \l__font_cjk_fallback_tl
              }
            \str_if_eq:VnT \l__font_layout_tl {vertical}
              {
                \str_if_eq:VnT \l__writing_selected_scope_tl {global}
                  {
                    \__writing_error:nn {Global~vertical~writing~is~unsupported}
                      {Use~a~local~Writing/Font~call~with~writing=vertical.}
                  }
              }
            \prop_map_inline:Nn \l__writing_overrides_prop
              { \__writing_assign_style:nn {##1} {##2} }
            \str_if_eq:VnT \l__writing_stylemap_tl {native}
              { \__writing_regular_fallbacks: }
            \bool_if:NT \l__writing_valid_bool
              {
                % Missing user-supplied resources are an error rather than
                % a silent change of font; old families retain soft fallback.
                \bool_lazy_or:nnT
                  { !\tl_if_blank_p:V \l__writing_font_tl }
                  { !\tl_if_blank_p:V \l__writing_file_tl }
                  { \tl_set:Nn \l__font_fallback_mode_tl {strict} }
                \__font_apply_declaration:
                \bool_if:NT \l__writing_local_call_bool
                  {
                    \tl_gset:NV \g__writing_last_local_command_tl \l__font_command_tl
                    \str_if_eq:VnT \l__font_layout_tl {vertical}
                      { \tl_gput_right:Nn \g__writing_last_local_command_tl {v} }
                  }
                \tl_if_blank:VF \l__writing_public_command_tl
                  {
                    \prop_gput:NVn \g__writing_owned_public_command_prop
                      \l__writing_public_command_tl {new-api}
                  }
                \tl_set:NV \l__font_fallback_mode_tl \l__writing_saved_fallback_tl
              }
          }
      }

% Legacy preset -> Unicode profile: old names stay valid, including
% sanskrit, hindi, sim, syriac_eastern, and the old local/global option.
% The registry loader remains the compatibility backend of the dispatcher:
% rewriting its routing would change old PDFs.
\cs_new_protected:Npn \__writing_legacy_preset:nn #1#2
  {
    \prop_if_in:NnTF \g__font_family_default_prop {#1}
      {
        \tl_if_blank:nTF {#2}
          { \__font_family_use_default:n {#1} }
          { \__font_family_use_loader:nn {#1} {#2} }
      }
      {
        % A new resource alias is a convenient UseFont preset as well.
        \prop_get:NnNTF \g__writing_legacy_profile_prop {#1} \l_tmpa_tl
          {
            \tl_if_blank:nTF {#2}
              {
                \tl_set:NV \l__writing_profile_tl \l_tmpa_tl
                \tl_set:Nn \l__writing_options_tl {#1}
                \__writing_apply:
              }
              {
                \PackageError{impe-writing}
                  {New~font~aliases~do~not~support~legacy~scope~overrides}
                  {Use~a~legacy~preset~for~[local|global],~or~select~a~font~in~UseWriting.}
              }
          }
          {
            \PackageError{impe-writing}{Unknown~font~preset~'#1'}
              {Select~a~registered~legacy~preset~or~new~font~alias.}
          }
      }
  }
