% Parse [optional pack, key=value, ...]profile.  TeX brace groups are
% restored when mapping; commas inside braced file names stay protected.
\cs_new_protected:Npn \__writing_entry_char:n #1
  {
    \str_if_eq:eeTF {\tl_to_str:n {#1}} {[}
      { \bool_set_true:N \l__writing_in_options_bool }
      {
        \str_if_eq:eeTF {\tl_to_str:n {#1}} {]}
          { \bool_set_false:N \l__writing_in_options_bool }
          {
            \bool_if:NTF \l__writing_in_options_bool
              { \tl_put_right:Nn \l__writing_options_tl {#1} }
              { \tl_put_right:Nn \l__writing_profile_tl {#1} }
          }
      }
  }
\cs_new_protected:Npn \__writing_parse_entry:n #1
  {
    \tl_clear:N \l__writing_options_tl
    \tl_clear:N \l__writing_profile_tl
    \bool_set_false:N \l__writing_in_options_bool
    \tl_map_inline:nn {#1}
      {
        \tl_if_single_token:nTF {##1}
          { \__writing_entry_char:n {##1} }
          {
            \bool_if:NTF \l__writing_in_options_bool
              { \tl_put_right:Nn \l__writing_options_tl { {##1} } }
              { \tl_put_right:Nn \l__writing_profile_tl { {##1} } }
          }
      }
    \tl_replace_all:Nnn \l__writing_profile_tl {\__writing_space_marker:} { ~ }
    \tl_replace_all:Nnn \l__writing_options_tl {\__writing_space_marker:} { ~ }
    \tl_trim_spaces:N \l__writing_profile_tl
    \tl_trim_spaces:N \l__writing_options_tl
    \tl_if_blank:VF \l__writing_profile_tl
      { \__writing_apply: }
  }

\cs_new_protected:Npn \__writing_item_char:n #1
  {
    \str_if_eq:eeTF {\tl_to_str:n {#1}} {,}
      {
        \int_compare:nNnTF {\l__writing_brackets_int} = {0}
          {
            \exp_args:NV \__writing_parse_entry:n \l__writing_entry_tl
            \tl_clear:N \l__writing_entry_tl
          }
          { \tl_put_right:Nn \l__writing_entry_tl {#1} }
      }
      {
        \str_if_eq:eeT {\tl_to_str:n {#1}} {[}
          { \int_incr:N \l__writing_brackets_int }
        \str_if_eq:eeT {\tl_to_str:n {#1}} {]}
          { \int_decr:N \l__writing_brackets_int }
        \tl_put_right:Nn \l__writing_entry_tl {#1}
      }
  }
\cs_new_protected:Npn \__writing_parse:n #1
  {
    % Undelimited TeX argument scanning skips category-10 spaces. Convert
    % them to an inert marker while scanning, then restore them before l3keys.
    \tl_set:Nn \l__writing_input_tl {#1}
    \tl_replace_all:Nnn \l__writing_input_tl { ~ } {\__writing_space_marker:}
    \tl_clear:N \l__writing_entry_tl
    \int_zero:N \l__writing_brackets_int
    \exp_args:NV \tl_map_inline:nn \l__writing_input_tl
      {
        \tl_if_single_token:nTF {##1}
          { \__writing_item_char:n {##1} }
          { \tl_put_right:Nn \l__writing_entry_tl { {##1} } }
      }
    \int_compare:nNnF {\l__writing_brackets_int} = {0}
      { \__writing_error:nn {Unbalanced~UseWriting~brackets} {Close~every~optional~[...].} }
    \exp_args:NV \__writing_parse_entry:n \l__writing_entry_tl
  }

\cs_new_protected:Npn \__writing_read_options:
  {
    \__writing_reset_options:
    \prop_get:NVNTF \g__writing_profile_pack_prop \l__writing_profile_tl \l__writing_pack_tl
      { }
      { \__writing_error:nn {Unknown~writing~profile~'\l__writing_profile_tl'} {Use~a~registered~profile.} }
    \tl_if_blank:VF \l__writing_options_tl
      {
        \exp_args:NNnV \seq_set_split:Nnn \l__writing_option_parts_seq {,} \l__writing_options_tl
        \seq_pop_left:NN \l__writing_option_parts_seq \l_tmpa_tl
        \tl_trim_spaces:N \l_tmpa_tl
        \tl_if_in:VnTF \l_tmpa_tl {=}
          { \seq_put_left:NV \l__writing_option_parts_seq \l_tmpa_tl }
          {
            \tl_if_blank:VF \l_tmpa_tl
              { \tl_set:NV \l__writing_pack_tl \l_tmpa_tl }
          }
        \tl_set:Nx \l_tmpa_tl { \seq_use:Nn \l__writing_option_parts_seq {,} }
        \tl_if_blank:VF \l_tmpa_tl
          { \exp_args:NnV \keys_set:nn {impe/writing} \l_tmpa_tl }
      }
    \bool_if:NT \l__writing_valid_bool
      {
        \prop_get:NVNTF \g__writing_pack_source_prop \l__writing_pack_tl \l__writing_source_tl
          { }
          { \__writing_error:nn {Unknown~font~pack~'\l__writing_pack_tl'} {Use~a~registered~pack~alias,~font=...,~or~file=....} }
      }
    \tl_if_blank:VF \l__writing_font_tl
      {
        \tl_if_blank:VF \l__writing_file_tl
          { \__writing_error:nn {font~and~file~are~mutually~exclusive} {Choose~one~resource~source.} }
      }
    \tl_if_blank:VTF \l__writing_stylemap_tl
      {
        \prop_get:NVNTF \g__writing_pack_default_stylemap_prop \l__writing_pack_tl \l__writing_stylemap_tl
          { }
          { \tl_set:Nn \l__writing_stylemap_tl {native} }
      }
      { }
    \str_if_eq:VnF \l__writing_stylemap_tl {native}
      {
        \str_if_eq:VnF \l__writing_stylemap_tl {full}
          { \__writing_error:nn {Invalid~stylemap} {Use~native~or~full.} }
      }
    \tl_if_blank:VF \l__writing_writing_tl
      {
        \str_if_eq:VnF \l__writing_writing_tl {horizontal}
          {
            \str_if_eq:VnF \l__writing_writing_tl {vertical}
              { \__writing_error:nn {Invalid~writing~mode} {Use~horizontal~or~vertical.} }
          }
      }
  }

            \tl_if_blank:VF \l__writing_writing_tl
              {
                \str_if_eq:VnTF \l__writing_writing_tl {horizontal}
                  { \tl_clear:N \l__font_layout_tl }
                  { \tl_set:Nn \l__font_layout_tl {vertical} }
              }
            \prop_map_inline:Nn \l__writing_overrides_prop
              { \__writing_assign_style:nn {##1} {##2} }
            \str_if_eq:VnT \l__writing_stylemap_tl {native}
              { \__writing_regular_fallbacks: }
            \bool_if:NT \l__writing_valid_bool
              {
                % Missing user-supplied resources are an error rather than
                % a silent change of font; old families retain soft fallback.
                \bool_lazy_or:nnT
                  { !\tl_if_blank_p:V \l__writing_font_tl }
                  { !\tl_if_blank_p:V \l__writing_file_tl }
                  { \tl_set:Nn \l__font_fallback_mode_tl {strict} }
                \__font_apply_declaration:
                \tl_set:NV \l__font_fallback_mode_tl \l__writing_saved_fallback_tl
              }
          }
      }
