\tl_new:N \g__test_selected_font_tl
\NewDocumentCommand\CheckFont{m}{
  \tl_set:Nx \l_tmpa_tl {\fontname\font}
  \exp_args:NNx \tl_if_in:NnF \l_tmpa_tl {\tl_to_str:n{#1}}
    {\errmessage{Wrong~font:~\l_tmpa_tl;~expected~#1}}
  \bool_if:NF \l__font_local_override_bool
    {\errmessage{Local~override~not~active}}
}

    {\errmessage{Generic~and~family~commands~differ}}}
