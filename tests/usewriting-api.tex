\typeout{IMPE-TEST-FULL-ITALIC:\tl_to_str:V \l__font_resolved_italic_tl}

  {\PackageError{impe-test}{Full~style~override~was~not~applied}{}}

\UseWriting{[shanggu, stylemap=full, italic=zhuque]chinese, mongolian}
% Without full, the unrelated Zhuque italic is deliberately absent.
\UseWriting{[shanggu]chinese}
\ExplSyntaxOn
\str_if_eq:eeF {\tl_use:N \l__font_resolved_italic_tl} {\tl_use:N \l__font_regular_tl}
  {\PackageError{impe-test}{Native~stylemap~must~fall~back~to~normal}{}}
\ExplSyntaxOff
% Empty pack uses the existing per-profile default; native faces include
% the declared same-series Shanggu bold and sans families.
\UseWriting{[stylemap=native]chinese}
\ExplSyntaxOn
\tl_if_blank:VTF \l__font_bold_tl
  {\PackageError{impe-test}{Native~Shanggu~bold~missing}{}}
  {}
\tl_if_blank:VTF \l__font_sans_tl
  {\PackageError{impe-test}{Native~Shanggu~sans~missing}{}}
  {}
\ExplSyntaxOff
